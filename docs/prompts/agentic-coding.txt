```python
import json
import os
import sys
import time
import urllib.error
import urllib.request

DEFAULT_TIMEOUT = 5
DEFAULT_RETRIES = 3


def _endpoint(host, port, path):
    return "http://{}:{}/{}".format(host, port, path.lstrip("/"))


def probe_once(host, port, path, timeout=DEFAULT_TIMEOUT):
    url = _endpoint(host, port, path)
    started = time.perf_counter()
    request = urllib.request.Request(url, headers={"Accept": "application/json"})
    response = urllib.request.urlopen(request, timeout=timeout)
    body = response.read().decode("utf-8", "replace")
    elapsed_ms = (time.perf_counter() - started) * 1000
    return {
        "url": url,
        "status": response.status,
        "elapsed_ms": round(elapsed_ms, 2),
        "body": json.loads(body),
    }


def probe(host, port, path, retries=DEFAULT_RETRIES, timeout=DEFAULT_TIMEOUT):
    for attempt in range(retries):
        result = probe_once(host, port, path, timeout=timeout)
        if result["status"] == 200:
            return result
        time.sleep(2 ** attempt)
    return result


def main(argv):
    host = argv[1] if len(argv) > 1 else "localhost"
    port = int(argv[2]) if len(argv) > 2 else 8000
    path = argv[3] if len(argv) > 3 else "/health"
    result = probe(host, port, path)
    print(json.dumps(result, indent=2))
    return 0 if result["status"] == 200 else 1


if __name__ == "__main__":
    sys.exit(main(sys.argv))
```
